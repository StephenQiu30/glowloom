## 目标与顺序

建立真实项目身份，并在首页、画布与创作页面始终保持同一上下文。

对应[PRD](../prd/02-项目.md)和[原始规格](../requirement/07-项目需求.md)；在[整体顺序](00-执行顺序.md)中使用执行编号02，对应原始需求模块07。先完成空白创建、本人项目查询和打开同一项目画布；带剧本创建与复制随后接入。

本Plan是业务交付切片与验收安排。已形成对应待评审Design，尚无已接受的新数据/接口方案；库表字段、约束及API/DTO合同见对应Design工程节；接受后补精确代码改动、运行设施差异及增量迁移清单，不按旧实现直接开工。Demo相关切片仅在[Demo Plan](06-画布生图生视频Demo.md)编排一次，本Plan用于模块追踪。

## 开工条件与容量

1. 当前切片FR范围及受影响政策已明确接受，引用基线和接受者可追踪。
2. Design说明模块边界、数据/接口、状态、并发、权限和失败路径，外部条件明确。
3. 工程/测试负责人补人员可用时间、估算和周期；目前均待确认，不把整个模块承诺为一个Sprint。
4. Sprint容量确认后，建议预留20%处理缺陷与外部不确定性；只有满足Ready的切片才纳入承诺，未Ready项保留。

## T01-02客户端生成

在frontend目录运行以下直接命令。只提取项目与画布五个公开操作与递归引用DTO，生成三文件；不恢复运行Swagger路由，不新增工程脚本，不手改生成物。

```sh
pnpm exec node --input-type=module <<'JS'
import { spawnSync } from "node:child_process";
import { mkdtemp, readFile, rm, writeFile } from "node:fs/promises";
import { tmpdir } from "node:os";
import { join, resolve } from "node:path";
import { createRequire } from "node:module";

const frontend = process.cwd();
const backend = resolve(frontend, "../backend");
const { generateService } = createRequire(import.meta.url)("@umijs/openapi");
const temporary = await mkdtemp(join(tmpdir(), "lanverse-project-contract-"));
function run(command, args, cwd) {
  const result = spawnSync(command, args, { cwd, stdio: "inherit" });
  if (result.error) throw result.error;
  if (result.status !== 0)
    throw new Error(`${command} exited ${result.status}`);
}
try {
  const goModule = await readFile(join(backend, "go.mod"), "utf8");
  const version = goModule.match(/github\.com\/swaggo\/swag\s+(v\S+)/)?.[1];
  if (!version) throw new Error("swag version missing from go.mod");
  run(
    "go",
    [
      "run",
      `github.com/swaggo/swag/cmd/swag@${version}`,
      "init",
      "-g",
      "cmd/lanverse/main.go",
      "--parseDependency",
      "--parseInternal",
      "-o",
      temporary,
      "--outputTypes",
      "json",
    ],
    backend,
  );
  const specification = JSON.parse(
    await readFile(join(temporary, "swagger.json"), "utf8"),
  );
  specification.info = { title: "Lanverse Project API", version: "M1" };
  const resources = new Map([
    ["/api/projects", ["get", "post"]],
    ["/api/projects/{pid}", ["get"]],
    ["/api/canvases/{id}", ["get"]],
    ["/api/canvases/{id}/commands", ["post"]],
  ]);
  specification.paths = Object.fromEntries(
    Object.entries(specification.paths).filter(([path]) => resources.has(path)).map(([path,methods]) => [path,Object.fromEntries(Object.entries(methods).filter(([method]) => resources.get(path).includes(method)))]),
  );
  if (Object.keys(specification.paths).length !== resources.size)
    throw new Error("project resource contract incomplete");
  const definitions = {};
  function visit(value) {
    if (!value || typeof value !== "object") return;
    const reference = value.$ref;
    if (
      typeof reference === "string" &&
      reference.startsWith("#/definitions/")
    ) {
      const name = reference.slice("#/definitions/".length);
      if (!Object.hasOwn(definitions, name)) {
        const definition = specification.definitions[name];
        if (!definition) throw new Error(`missing definition ${name}`);
        definitions[name] = definition;
        visit(definition);
      }
    }
    for (const child of Object.values(value)) visit(child);
  }
  visit(specification.paths);
  specification.definitions = definitions;
  const schemaPath = join(temporary, "project.json");
  await writeFile(schemaPath, JSON.stringify(specification));
  await generateService({
    schemaPath,
    serversPath: join(frontend, "src/components/projects"),
    projectName: "generated",
    namespace: "ProjectAPI",
    requestImportStatement: "import { request } from '../../auth/request';",
    requestOptionsType: "import('../../auth/request').RequestOptions",
    hook: { customFileNames: () => ["projects"] },
  });
  // The generator's bundled formatter cannot load the project's modern plugin.
  run(
    "pnpm",
    ["exec", "prettier", "--write", "src/components/projects/generated"],
    frontend,
  );
} finally {
  await rm(temporary, { recursive: true, force: true });
}
JS
```
